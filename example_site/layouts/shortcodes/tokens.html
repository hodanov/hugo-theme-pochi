{{- /* Token swatches for the styleguide page (example_site only, not part of the theme). Reads the theme's tokens.css and renders every --pochi-* custom property of the main :root block: primitive colors (OKLCH), semantic colors (anything that resolves to a color, shown with the primitives it picks per theme), shadows and the rest. Swatches use var() so they follow the active theme. */ -}} {{- $css := (resources.Get "css/tokens.css").Content -}} {{- $root := index (split $css "\n}\n") 0 -}} {{- $decls := findRESubmatch `(--pochi-[a-z0-9-]+)\s*:\s*([^;]+);` $root -}} {{- $seen := slice -}} {{- $colorNames := slice -}} {{- $primitives := slice -}} {{- $semantic := slice -}} {{- $shadows := slice -}} {{- $others := slice -}} {{- range $decls -}} {{- $name := index . 1 -}} {{- $value := replaceRE `\s+` " " (index . 2) | replaceRE `\(\s+` "(" | replaceRE `\s+\)` ")" | strings.TrimSpace -}} {{- $ref := replaceRE `^var\((--pochi-[a-z0-9-]+)\)$` "$1" $value -}} {{- if not (in $seen $name) -}} {{- $seen = $seen | append $name -}} {{- $item := dict "name" $name "value" (replaceRE `var\(--pochi-([a-z0-9-]+)\)` "$1" $value) -}} {{- if hasPrefix $name "--pochi-shadow-" -}} {{- $shadows = $shadows | append $name -}} {{- else if hasPrefix $value "oklch(" -}} {{- $colorNames = $colorNames | append $name -}} {{- $primitives = $primitives | append $item -}} {{- else if or (findRE `^(light-dark|rgba?|color-mix)\(` $value) (in $colorNames $ref) -}} {{- $colorNames = $colorNames | append $name -}} {{- $semantic = $semantic | append $item -}} {{- else -}} {{- $others = $others | append $item -}} {{- end -}} {{- end -}} {{- end -}}
{{- range slice (dict "title" "Color: primitive" "items" $primitives) (dict "title" "Color: semantic" "items" $semantic) }}

    {{- range .items }}
  • {{ .name }}
    {{ .value }}
    • {{- end }}
{{- end }}

  • {{ . }}
