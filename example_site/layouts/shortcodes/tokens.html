{{- /* Token swatches for the styleguide page (example_site only, not part of the theme). Reads the theme's tokens.css and renders every --pochi-* custom property declared before the dark-mode block. Swatches use var() so they follow the active theme; values shown are the light-mode declarations. */ -}} {{- $css := (resources.Get "css/tokens.css").Content -}} {{- $light := index (split $css ":root.dark") 0 -}} {{- $decls := findRESubmatch `(--pochi-[a-z0-9-]+)\s*:\s*([^;]+);` $light -}} {{- $seen := slice -}} {{- $colors := slice -}} {{- $shadows := slice -}} {{- $others := slice -}} {{- range $decls -}} {{- $name := index . 1 -}} {{- $value := replaceRE `\s+` " " (index . 2) | strings.TrimSpace -}} {{- if not (in $seen $name) -}} {{- $seen = $seen | append $name -}} {{- if in $name "shadow" -}} {{- $shadows = $shadows | append $name -}} {{- else if findRE `^(#|rgba?\(|var\()` $value -}} {{- $colors = $colors | append (dict "name" $name "value" $value) -}} {{- else -}} {{- $others = $others | append (dict "name" $name "value" $value) -}} {{- end -}} {{- end -}} {{- end -}}

    {{- range $colors }}
  • {{ .name }}
    • {{- end }}

Shadow

    {{- range $shadows }}
  • {{ . }}
    • {{- end }}

Other

    {{- range $others }}
  • {{ .name }}: {{ .value }}
    • {{- end }}
